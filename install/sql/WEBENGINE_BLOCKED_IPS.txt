DROP TABLE IF EXISTS `webengine_blocked_ips`;

CREATE TABLE `webengine_blocked_ips` (
  `ip_address` VARCHAR(45) NOT NULL PRIMARY KEY,         -- Dirección IP
  `fail_count` INT NOT NULL DEFAULT 0,                   -- Intentos fallidos actuales
  `ban_count` INT NOT NULL DEFAULT 0,                    -- Cantidad de bloqueos previos (temporales)
  `block_until` DATETIME DEFAULT NULL,                   -- Hasta cuándo está bloqueado (null = no bloqueado)
  `reason` TEXT DEFAULT NULL,                            -- Motivo del bloqueo
  `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP -- Último intento fallido
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
