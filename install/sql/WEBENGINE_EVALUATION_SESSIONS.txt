CREATE TABLE webengine_evaluation_sessions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    project_id INT NOT NULL,
    creado_por INT NOT NULL,
    estado ENUM('activa', 'cerrada') NOT NULL DEFAULT 'activa',
    token_acceso CHAR(64) NOT NULL UNIQUE,
    inicio DATETIME NOT NULL,
    duracion INT UNSIGNED NOT NULL,
    fin DATETIME GENERATED ALWAYS AS (inicio + INTERVAL duracion SECOND) STORED,
    cierre_manual BOOLEAN DEFAULT FALSE,
    cierre_motivo ENUM('expirada','manual','error') DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (project_id) REFERENCES webengine_projects(id) ON DELETE RESTRICT,
    FOREIGN KEY (creado_por) REFERENCES webengine_u_core(uid) ON DELETE RESTRICT,
    INDEX idx_project_estado (project_id, estado),
    INDEX idx_inicio (inicio),
    INDEX idx_token (token_acceso)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
