CREATE TABLE webengine_projects (
    id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(255) NOT NULL,
    linea_investigacion_id INT NOT NULL,
    visibilidad ENUM('publico', 'privado') DEFAULT 'privado',
    activo TINYINT(1) DEFAULT 1,
    directorio VARCHAR(255),
    
    version VARCHAR(10) NOT NULL DEFAULT '1.0',
    
    fase ENUM('propuesta', 'desarrollo', 'aplicacion') DEFAULT 'propuesta',
    estado ENUM('nuevo', 'en_evaluacion', 'completado') DEFAULT 'nuevo',
    
    descripcion TEXT,
    palabras_clave VARCHAR(255),
    
    calificado TINYINT(1) DEFAULT 0,
    puntuacion DECIMAL(3,1),
    
    timer_segundos INT NOT NULL DEFAULT 0,
    hora_programada TIME,
    fecha_presentacion DATE COMMENT 'Fecha de presentación del proyecto',
    
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_linea_investigacion (linea_investigacion_id),
    INDEX idx_estado (estado),
    INDEX idx_fase (fase),
    
    FOREIGN KEY (linea_investigacion_id) REFERENCES webengine_research_lines(id) 
        ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
