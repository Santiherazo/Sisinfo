CREATE TABLE webengine_project_teachers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    project_id INT NOT NULL,
    usuario_uid INT NOT NULL,
    rol ENUM('director', 'jurado', 'asesor') DEFAULT 'asesor',
    estado ENUM('activo','removido') DEFAULT 'activo',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (project_id, usuario_uid),
    FOREIGN KEY (project_id) REFERENCES webengine_projects(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_uid) REFERENCES webengine_u_core(uid) ON DELETE CASCADE
);
