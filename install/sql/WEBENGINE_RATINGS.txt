CREATE TABLE webengine_ratings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    session_token VARCHAR(64) NOT NULL COMMENT 'Identificador único de la sesión de evaluación',
    project_id INT NOT NULL,
    evaluador_uid INT NOT NULL,
    criterio_nombre VARCHAR(255) NOT NULL COMMENT 'Nombre del criterio evaluado',
    criterio_valor TEXT COMMENT 'Valor/criterio específico del formulario',
    calificacion DECIMAL(3,1) NOT NULL,
    observacion_personal TEXT,
    estado ENUM('evaluado', 'pendiente', 'corregir') DEFAULT 'pendiente' COMMENT 'Estado de la evaluación',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY (project_id, evaluador_uid, criterio_nombre) COMMENT 'Un evaluador no puede calificar dos veces el mismo criterio',
    FOREIGN KEY (project_id) REFERENCES webengine_projects(id) ON DELETE CASCADE,
    FOREIGN KEY (evaluador_uid) REFERENCES webengine_u_core(uid) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
