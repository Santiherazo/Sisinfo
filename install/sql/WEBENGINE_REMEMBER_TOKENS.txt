DROP TABLE IF EXISTS webengine_remember_tokens;

CREATE TABLE webengine_remember_tokens (
    token_id CHAR(64) PRIMARY KEY COMMENT 'Token de autenticación persistente (ej. SHA-256)',
    uid INT NOT NULL COMMENT 'ID del usuario relacionado',
    ip_address VARCHAR(45) DEFAULT NULL COMMENT 'Última IP usada con este token',
    user_agent VARCHAR(512) DEFAULT NULL COMMENT 'User-Agent del navegador del cliente',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de creación del token',
    expires_at DATETIME NOT NULL COMMENT 'Fecha de expiración del token (ej. 30 días)',
    last_used_at DATETIME DEFAULT NULL COMMENT 'Última vez que se usó este token',

    CONSTRAINT fk_token_user FOREIGN KEY (uid) REFERENCES webengine_u_core(uid) ON DELETE CASCADE,

    INDEX idx_token_uid (uid),
    INDEX idx_token_expiry (expires_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Tokens persistentes para recordar inicio de sesión';
