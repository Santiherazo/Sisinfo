    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL,
    description TEXT,
    estado BOOLEAN DEFAULT TRUE COMMENT 'Estado del rol: TRUE=Activo, FALSE=Inactivo'
);
