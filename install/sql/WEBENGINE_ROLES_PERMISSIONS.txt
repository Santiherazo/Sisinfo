CREATE TABLE IF NOT EXISTS webengine_role_permissions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    role_id INT NOT NULL,
    permission_id INT NOT NULL,
    UNIQUE (role_id, permission_id),
    FOREIGN KEY (role_id) REFERENCES webengine_roles(id) ON DELETE CASCADE,
    FOREIGN KEY (permission_id) REFERENCES webengine_permissions(id) ON DELETE CASCADE
);

INSERT IGNORE INTO webengine_role_permissions (role_id, permission_id)
SELECT 1, id FROM webengine_permissions;

INSERT IGNORE INTO webengine_role_permissions (role_id, permission_id)
SELECT 2, id FROM webengine_permissions 
WHERE (module = 'Cpuser' AND action IN ('create', 'manage', 'manager'))
   OR (module = 'Cpproject' AND action IN ('create', 'manage', 'manager'))
   OR (module = 'lineManager' AND action = 'manage')
   OR (module = 'Estadisticas' AND action = 'ver')
   OR (module = 'usercp' AND action = 'results')
   OR (module = 'Cpanel' AND action = 'access')
   OR (module = 'Cpeval' AND action = 'manager');

INSERT IGNORE INTO webengine_role_permissions (role_id, permission_id)
SELECT 3, id FROM webengine_permissions 
WHERE module = 'Epanel' AND action = 'access';

INSERT IGNORE INTO webengine_role_permissions (role_id, permission_id)
SELECT 4, id FROM webengine_permissions 
WHERE (module = 'usercp' AND action = 'projects') 
   OR (module = 'usercp' AND action = 'results');

INSERT IGNORE INTO webengine_role_permissions (role_id, permission_id)
SELECT 5, id FROM webengine_permissions 
WHERE (module = 'usercp' AND action = 'projects') 
   OR (module = 'usercp' AND action = 'results');

SELECT r.name as rol, p.module, p.action, p.description
FROM webengine_role_permissions rp
JOIN webengine_roles r ON rp.role_id = r.id
JOIN webengine_permissions p ON rp.permission_id = p.id
WHERE r.name IN ('docente', 'estudiante')
AND p.module = 'usercp'
ORDER BY r.name, p.action;
