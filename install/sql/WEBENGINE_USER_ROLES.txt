CREATE TABLE webengine_user_roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    role_id INT NOT NULL,
    UNIQUE (user_id, role_id),
    FOREIGN KEY (user_id) REFERENCES webengine_u_core(uid) ON DELETE CASCADE,
    FOREIGN KEY (role_id) REFERENCES webengine_roles(id) ON DELETE CASCADE
);
